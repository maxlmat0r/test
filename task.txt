
#!/bin/bash
#
#
# Локальный сбор метрики "Process schedule stats CPU Waiting (MAX)" напрямую
# из первоисточника /proc/schedstat, без Prometheus/node_exporter.
#
# Аналог PromQL-запроса:
#   max(irate(node_schedstat_waiting_seconds_total{instance=~"$node"}[$interval])) by (instance)
#
# Как это устроено:
#   - node_exporter берёт wait_time из /proc/schedstat: строка "cpuN f1 f2 ... f7 run_time wait_time timeslices"
#     (9 полей после имени CPU; wait_time — 8-е из них, в наносекундах).
#   - node_schedstat_waiting_seconds_total = wait_time / 1e9 (счётчик в секундах).
#   - irate() = (current - previous) / (time_current - time_previous) по двум последним точкам.
#   - max(...) by (instance) = берём максимум среди всех CPU хоста.
#
# Скрипт делает ровно то же самое локально: на каждой итерации считает дельту
# wait_time по каждому cpuN между двумя соседними снятиями, делит на реальный
# прошедший интервал времени, и берёт максимум по всем CPU.
#
# Запуск:
#   chmod +x cpuwait.sh
#   ./cpuwait.sh
#   (или в фоне: nohup ./cpuwait.sh > /tmp/cpuwait.log 2>&1 &)
#
# Результат: /tmp/cpuwait.txt
#

set -uo pipefail

OUTFILE="/tmp/cpuwait.txt"
DURATION_SEC=$((30 * 60))   # 30 минут
INTERVAL_SEC=5             # интервал снятия метрики
SCHEDSTAT="/proc/schedstat"

if [ ! -r "$SCHEDSTAT" ]; then
    echo "Ошибка: $SCHEDSTAT недоступен для чтения" >&2
    exit 1
fi

: > "$OUTFILE"

# --- Функция чтения текущего wait_time (нс) по каждому CPU из /proc/schedstat ---
# Формат строки: cpu0 <f1> <f2> <f3> <f4> <f5> <f6> <run_time> <wait_time> <timeslices>
# awk: $1=cpuN, $9=wait_time (9-е поле после имени CPU в столбцах $2..$10)
read_schedstat() {
    awk '/^cpu[0-9]+ / {print $1, $9}' "$SCHEDSTAT"
}

# --- Форматирование значения: мс если < 1с, иначе секунды ---
format_value() {
    local val="$1"
    awk -v v="$val" 'BEGIN {
        if (v < 1) {
            printf "%.2f ms", v * 1000
        } else {
            printf "%.3f s", v
        }
    }'
}

# --- Сравнение двух float-чисел (a > b -> 1, иначе 0) ---
is_greater() {
    awk -v a="$1" -v b="$2" 'BEGIN{ print (a > b) ? 1 : 0 }'
}

echo "Сбор метрики CPU Waiting запущен. Интервал: ${INTERVAL_SEC}s, длительность: $((DURATION_SEC/60)) мин."
echo "Отчёт: $OUTFILE"

# --- Начальный снимок ---
declare -A prev_wait
while read -r cpu wait_ns; do
    prev_wait["$cpu"]="$wait_ns"
done < <(read_schedstat)

prev_time=$(date +%s.%N)
end_time=$(( $(date +%s) + DURATION_SEC ))

min_val=""
max_val=""

while [ "$(date +%s)" -lt "$end_time" ]; do
    sleep "$INTERVAL_SEC"

    curr_time=$(date +%s.%N)
    dt=$(awk -v a="$curr_time" -v b="$prev_time" 'BEGIN{ printf "%.6f", a-b }')

    max_rate="0"
    while read -r cpu wait_ns; do
        prev="${prev_wait[$cpu]:-$wait_ns}"
        delta_ns=$(( wait_ns - prev ))

        # защита от отрицательной дельты (переполнение счётчика/рестарт) — считаем как 0
        if [ "$delta_ns" -lt 0 ]; then
            delta_ns=0
        fi

        rate=$(awk -v d="$delta_ns" -v t="$dt" 'BEGIN{
            if (t > 0) { printf "%.6f", (d / 1000000000) / t } else { print 0 }
        }')

        if [ "$(is_greater "$rate" "$max_rate")" -eq 1 ]; then
            max_rate="$rate"
        fi

        prev_wait["$cpu"]="$wait_ns"
    done < <(read_schedstat)

    prev_time="$curr_time"

    ts=$(date '+%Y-%m-%d %H:%M:%S')
    formatted=$(format_value "$max_rate")
    echo "${ts} - ${formatted}" >> "$OUTFILE"

    # обновление min/max
    if [ -z "$min_val" ] || [ "$(is_greater "$min_val" "$max_rate")" -eq 1 ]; then
        min_val="$max_rate"
    fi
    if [ -z "$max_val" ] || [ "$(is_greater "$max_rate" "$max_val")" -eq 1 ]; then
        max_val="$max_rate"
    fi
done

{
    echo ""
    echo "=== Итог за $((DURATION_SEC/60)) минут (интервал ${INTERVAL_SEC}s) ==="
    echo "Min: $(format_value "$min_val")"
    echo "Max: $(format_value "$max_val")"
} >> "$OUTFILE"

echo "Готово. Отчёт сохранён в $OUTFILE"
